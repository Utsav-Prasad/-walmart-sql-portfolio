TOP STORES BY REVENUE RANK

Using RANK() and DENSE_RANK() window functions to identify top-performing stores per region. Enables regional managers to benchmark against peers and flag underperformers.

01_store_revenue_ranking.sql ✓ 45 rows
-- Top stores by revenue with regional ranking WITH store_revenue AS ( SELECT s.store_id, s.store_name, s.region, s.city, SUM(f.total_amount) AS total_revenue, COUNT(f.sale_id) AS transaction_count, ROUND(AVG(f.total_amount),2) AS avg_basket_size FROM fact_sales f JOIN dim_stores s ON f.store_id = s.store_id GROUP BY s.store_id, s.store_name, s.region, s.city ), ranked_stores AS ( SELECT *, RANK() OVER ( PARTITION BY region ORDER BY total_revenue DESC ) AS region_rank, ROUND( total_revenue / SUM(total_revenue) OVER (PARTITION BY region) * 100, 2 ) AS region_share_pct FROM store_revenue ) SELECT region, store_name, city, FORMAT(total_revenue, 0) AS revenue_fmt, transaction_count, avg_basket_size, region_rank, CONCAT(region_share_pct, '%') AS region_share FROM ranked_stores WHERE region_rank <= 5 ORDER BY region, region_rank;
// query 02 — customer analytics
RFM CUSTOMER SEGMENTATION

Recency, Frequency, and Monetary scoring to classify customers into Champion, Loyal, At-Risk, and Churned segments — enabling targeted marketing campaigns.

02_rfm_segmentation.sql ✓ 84,392 rows
-- RFM Segmentation: Recency, Frequency, Monetary WITH rfm_base AS ( SELECT c.customer_id, c.full_name, c.member_type, DATEDIFF(CURDATE(), MAX(d.full_date)) AS recency_days, COUNT(f.sale_id) AS frequency, ROUND(SUM(f.total_amount), 2) AS monetary FROM fact_sales f JOIN dim_customers c ON f.customer_id = c.customer_id JOIN dim_date d ON f.date_id = d.date_id GROUP BY c.customer_id, c.full_name, c.member_type ), rfm_scores AS ( SELECT *, NTILE(5) OVER (ORDER BY recency_days ASC) AS r_score, NTILE(5) OVER (ORDER BY frequency DESC) AS f_score, NTILE(5) OVER (ORDER BY monetary DESC) AS m_score FROM rfm_base ) SELECT customer_id, full_name, member_type, recency_days, frequency, monetary, CONCAT(r_score, f_score, m_score) AS rfm_cell, CASE WHEN r_score >= 4 AND f_score >= 4 THEN 'Champion' WHEN r_score >= 3 AND f_score >= 3 THEN 'Loyal Customer' WHEN r_score >= 4 AND f_score <= 2 THEN 'New Customer' WHEN r_score <= 2 AND f_score >= 3 THEN 'At Risk' WHEN r_score <= 2 AND f_score <= 2 THEN 'Churned' ELSE 'Potential Loyalist' END AS customer_segment FROM rfm_scores ORDER BY monetary DESC;
// query 03 — trend analysis
MONTH-OVER-MONTH GROWTH

LAG() window function to compare each month's revenue against the prior month, highlighting seasonal spikes and decline periods.

03_mom_growth.sql ✓ 6 rows
WITH monthly AS ( SELECT d.month_name, d.quarter, YEAR(d.full_date) AS yr, MONTH(d.full_date) AS mo, SUM(f.total_amount) AS revenue FROM fact_sales f JOIN dim_date d ON f.date_id = d.date_id GROUP BY 1,2,3,4 ) SELECT month_name, yr, quarter, FORMAT(revenue, 0) AS revenue, FORMAT(LAG(revenue) OVER ( ORDER BY yr, mo), 0) AS prev_month, ROUND( (revenue - LAG(revenue) OVER (ORDER BY yr, mo)) / LAG(revenue) OVER (ORDER BY yr, mo) * 100, 2 ) AS growth_pct FROM monthly ORDER BY yr, mo;
// query 04 — inventory ops
SLOW-MOVING INVENTORY

Identify products with high shelf-life remaining but low sell-through rate — flagging clearance candidates before expiry losses occur.

04_slow_inventory.sql ✓ 312 rows
SELECT p.product_name, p.category, p.department, p.shelf_life_days, SUM(f.quantity) AS units_sold, ROUND( SUM(f.quantity) / p.shelf_life_days, 2 ) AS daily_velocity, CASE WHEN SUM(f.quantity) / p.shelf_life_days < 0.5 THEN '⚠ Clearance Risk' WHEN SUM(f.quantity) / p.shelf_life_days < 1.5 THEN 'Monitor' ELSE 'Healthy' END AS inventory_status FROM fact_sales f JOIN dim_products p ON f.product_id = p.product_id WHERE p.shelf_life_days IS NOT NULL GROUP BY p.product_id HAVING inventory_status != 'Healthy' ORDER BY daily_velocity ASC LIMIT 50;

AUTOMATED DAILY KPI REPORT

A stored procedure that runs nightly to populate a KPI summary table, enabling dashboards to read pre-aggregated results without hitting fact tables at query time.

12_store_proc_daily_kpi.sql ✓ Procedure Created
DELIMITER $$ CREATE PROCEDURE sp_generate_daily_kpi( IN p_date DATE, OUT p_total_rev DECIMAL(15,2), OUT p_txn_count INT ) BEGIN -- Clear existing data for this date DELETE FROM kpi_daily_summary WHERE report_date = p_date; -- Insert fresh aggregates INSERT INTO kpi_daily_summary ( report_date, store_id, total_revenue, transaction_count, avg_basket, unique_customers ) SELECT p_date, f.store_id, SUM(f.total_amount), COUNT(f.sale_id), AVG(f.total_amount), COUNT(DISTINCT f.customer_id) FROM fact_sales f JOIN dim_date d ON f.date_id = d.date_id WHERE d.full_date = p_date GROUP BY f.store_id; -- Return summary totals via OUT params SELECT SUM(total_revenue), SUM(transaction_count) INTO p_total_rev, p_txn_count FROM kpi_daily_summary WHERE report_date = p_date; END$$ DELIMITER ; -- Execute: CALL sp_generate_daily_kpi( '2024-06-30', @rev, @txn ); SELECT @rev AS daily_revenue, @txn AS daily_txns;
